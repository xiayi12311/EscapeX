local nowPawn={}; local pc={locks=0}; local timers={}; local lastRotation
function pc:SetIgnoreLookInput(on) self.locks=self.locks+(on and 1 or -1); assert(self.locks>=0) end
KismetSystemLibrary={IsValid=function(x) return x and not x.destroyed end}
UGCGameSystem={GetLocalPlayerController=function() return pc end,GetLocalPlayerPawn=function() return nowPawn end}
UGCTimerUtility={CreateLuaTimer=function(delay,fn) local t={fn=fn}; timers[#timers+1]=t; return t end,
    RemoveLuaTimer=function(t) t.removed=true end}
UGCPlayerControllerSystem={SetControlRotation=function(_,r) lastRotation=r end}
ugcprint=function() end
local C=dofile(TEST_ROOT..'/Script/Gameplay/Capture/CaptureCamera.lua')
local monster={yaw=40,K2_GetActorRotation=function(self) return {Yaw=self.yaw} end}
local post={yaw=90,K2_GetActorRotation=monster.K2_GetActorRotation}
C.Start(monster,'carry'); assert(pc.locks==1 and lastRotation.Yaw==220 and lastRotation.Pitch==-50)
C.Start(monster,'carry'); assert(pc.locks==1 and #timers==1)
monster.yaw=65; timers[1].fn(); assert(lastRotation.Yaw==245)
C.Start(post,'bind'); assert(timers[1].removed and pc.locks==1 and lastRotation.Yaw==90 and lastRotation.Pitch==0)
C.Stop(monster); assert(pc.locks==1)
nowPawn={}; timers[2].fn(); assert(timers[2].removed and pc.locks==0)
C.Start(monster,'carry'); monster.K2_GetActorRotation=function() error('bad rotation') end
timers[3].fn(); assert(timers[3].removed and pc.locks==0)
print('PASS capture camera follows target and balances look locks on transition and errors')
