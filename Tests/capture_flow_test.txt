local function requireModule(name)
    if name=='Script.Gameplay.Capture.CaptureCamera' then
        return {Start=function() end, Stop=function() end}
    end
    if cache[name] then return cache[name] end
    local path=TEST_ROOT..'/'..name:gsub('%.','/')..'.lua'
    local mod=dofile(path)
    assert(mod~=nil, 'module returned nil: '..name)
    cache[name]=mod
    return mod
end

local function bb()
    local b={}
    function b:GetValueAsObject(k) return self[k] end
    function b:SetValueAsObject(k,v) self[k]=v end
    function b:GetValueAsBool(k) return self[k] or false end
    function b:SetValueAsBool(k,v) self[k]=v end
    function b:GetValueAsInt(k) return self[k] or 0 end
    function b:SetValueAsInt(k,v) self[k]=v end
    function b:ClearValue(k) self[k]=nil end
    return b
end

local function playerPawn(key, loc)
    local p={key=key, position=loc or vec(50,0,88), health=100, name='Pawn'..key}
    p.CapsuleComponent=capsule(p,34,88)
    function p:HasAuthority() return true end
    function p:K2_GetActorLocation() return self.position end
    function p:GetActorBounds(_,c,e) local cc=self.boundsCenter or self.position
        local ee=self.boundsExtent or vec(34,34,88)
        c.X,c.Y,c.Z=cc.X,cc.Y,cc.Z; e.X,e.Y,e.Z=ee.X,ee.Y,ee.Z end
    function p:K2_AttachToActor(actor,socket) calls.attach={actor=actor,socket=socket}; return true end
    function p:K2_AttachToComponent(comp) calls.attachComponent=comp; return true end
    function p:K2_DetachFromActor() calls.detach=(calls.detach or 0)+1; return true end
    function p:K2_SetActorLocation(loc,sweep,hit,teleport) calls.place=loc; calls.placeTeleport=teleport; return true end
    function p:K2_SetActorRelativeLocation(loc,sweep,hit,teleport) calls.relative=loc; return true end
    p.CharacterMovement = {
        MovementMode = 1,
        tick = true,
        IsComponentTickEnabled = function(self) return self.tick end,
        SetComponentTickEnabled = function(self, enabled) self.tick = enabled end,
        SetMovementMode = function(self, mode)
            self.MovementMode = mode
            calls.modes = calls.modes or {}
            calls.modes[#calls.modes + 1] = mode
            return true
        end,
    }
    p.controller={key=key}
    pawns[key]=p
    return p
end

local function newMonster(loc)
    local m={position=loc or vec(0,0,100), name='Monster', bb=bb(), key=-1}
    m.Mesh={DoesSocketExist=function(_,socket) return socket=='headSocket' end,
        GetSocketLocation=function() return vec(m.position.X, m.position.Y, m.position.Z+150) end}
    m.CapsuleComponent=capsule(m,60,120)
    function m:HasAuthority() return true end
    function m:K2_GetActorLocation() return self.position end
    function m:HandleCaughtPlayer(target, seq, controller)
        m.handlerCalls=(m.handlerCalls or 0)+1
        return Capture.begin(m, target)
    end
    return m
end

    a.RescueArea={SetSphereRadius=function() end}

end

local function task(module)
    local t=setmetatable({}, {__index=module})
    function t:FinishExecute(ok) self.finished=ok end
    function t:FinishAbort() self.aborted=true end
    return t
end

-- walk the monster toward its target so arrival can happen without navigation
local function walk(monster, dt)
    local target=monster.lastMove
    if target and monster.moving then
        local dx,dy=target.X-monster.position.X, target.Y-monster.position.Y
        local len=math.sqrt(dx*dx+dy*dy)
        local step=(monster.speed or 300)*dt
        if len<=step or len==0 then monster.position=vec(target.X,target.Y,monster.position.Z)
        else monster.position=vec(monster.position.X+dx/len*step, monster.position.Y+dy/len*step, monster.position.Z) end
    end
end

-- walk the manager through the phases in the order the monster would report them
local function imprison(g, sessionId, postId)
    clock=clock+H.Defaults.HTPickupDuration+0.1
    g.CaptureManager:Update()                       -- Grabbing -> Carrying
    assert(g.CaptureManager:GetSnapshot(sessionId).State=='Carrying')
    assert(g.CaptureManager:ClaimPost(sessionId, postId))
    assert(g.CaptureManager:MarkArrived(sessionId))
    clock=clock+0.1
    g.CaptureManager:Update()                       -- Carrying -> PuttingDown
    clock=clock+H.Defaults.HTPutDownDuration+0.1
    g.CaptureManager:Update()                       -- PuttingDown -> Imprisoned
    local snap=g.CaptureManager:GetSnapshot(sessionId)
    assert(snap and snap.State=='Imprisoned', 'imprisoned, got '..tostring(snap and snap.State))
    return snap
end

check('movement simulation is restored on every server-side carry exit', function()
    local reasons={'PLAYER_DEAD','PLAYER_UNAVAILABLE','BT_ABORTED','MONSTER_DESTROYED','SERVER_SHUTDOWN','POST_REMOVED'}
    for _,reason in ipairs(reasons) do
        local g=freshWorld()
        local post=newPost('EXIT_POST',400,0,0)
        assert(Service.RegisterPost(g,post))
        local p=playerPawn(1)
        local monster=newMonster()
        local ok,sid=Service.StartCapture(g,monster,p)
        assert(ok)
        clock=clock+0.7; g.CaptureManager:Update()
        assert(p.CharacterMovement.MovementMode==0 and not p.CharacterMovement.tick, reason..': lock not acquired')
        if reason=='SERVER_SHUTDOWN' then Service.Shutdown(g)
        elseif reason=='POST_REMOVED' then
            imprison(g,sid,'EXIT_POST')
            assert(p.CharacterMovement.MovementMode==0 and not p.CharacterMovement.tick)
            assert(Service.UnregisterPost(g,post))
        elseif reason=='BT_ABORTED' or reason=='MONSTER_DESTROYED' then
            assert(Service.CancelMonster(g,monster,reason))
        else assert(Service.CancelPlayer(g,1,reason)) end
        assert(p.CharacterMovement.MovementMode==1 and p.CharacterMovement.tick, reason..': movement not restored')
    end
end)

check('movement restores after failed carry and bind attachments', function()
    for _,kind in ipairs({'carry','bind'}) do
        local g=freshWorld()
        assert(Service.RegisterPost(g,newPost('FAIL_POST',400,0,0)))
        local p=playerPawn(1)
        local monster=newMonster()
        local ok,sid=Service.StartCapture(g,monster,p)
        assert(ok)
        if kind=='carry' then p.K2_AttachToActor=function() return false end end
        clock=clock+0.7; g.CaptureManager:Update()
        if kind=='bind' then
            assert(p.CharacterMovement.MovementMode==0 and not p.CharacterMovement.tick)
            p.K2_AttachToComponent=function() return false end
            assert(Service.ClaimPost(g,sid,'FAIL_POST'))
            assert(Service.MarkArrived(g,sid))
            clock=clock+0.1; g.CaptureManager:Update()
            clock=clock+0.7; g.CaptureManager:Update()
        end
        assert(not g.CaptureManager:GetSnapshot(sid))
        assert(p.CharacterMovement.MovementMode==1 and p.CharacterMovement.tick, kind..' failure kept movement locked')
    end
end)

-- ---------- the full escort, end to end --------------------------------------
check('catch stays Running, escorts, binds and then finishes', function()
    local g=freshWorld()
    assert(Service.RegisterPost(g,newPost('NEAR',400,0,0)))
    assert(Service.RegisterPost(g,newPost('FAR',9000,0,0)))
    local m=newMonster()
    local target=playerPawn(1, vec(50,0,88))
    local Catch=requireModule('Script.AI.HeadTeacher.BTT_HTCatch')
    local t=task(Catch)
    m.position=vec(0,0,88)
    m.bb:SetValueAsObject('LockedTarget', target)
    m.bb:SetValueAsBool('CatchReady', true)
    m.bb:SetValueAsInt('ChaseSequenceId', 5)
    local d=H.data(m); d.catchGap=20; d.catchSeq=5
    KismetSystemLibrary.LineTraceSingleByCollisionChannel=function() return false end

    t:ReceiveExecuteAI({name='Controller'}, m)
    assert(t.finished==nil, 'catch must stay Running once a capture starts')
    assert(t.Capturing==true)
    assert(m.bb:GetValueAsObject('LockedTarget')==target, 'lock must survive while escorting')
    assert(m.handlerCalls==1)
    local cap=H.data(m).capture
    assert(cap and cap.id, 'capture session recorded')
    -- Grabbing: the player is already frozen and invincible (the pickup window must not
    -- stay walkable), but nothing is attached until the carry phase.
    assert(calls.attach==nil, 'attach must wait for the carry phase')
    assert(calls.entered==1 and calls.invincible==true, 'freeze starts at the grab itself')
    assert(m.HTCaptureState.Stage=='Grabbing' and m.HTCaptureState.PlayerKey==1,
        'grab stage must reach the local camera before the carry phase')
    assert((m.stops or 0)>=1)

    clock=clock+0.7
    g.CaptureManager:Update()
    t:ReceiveTickAI(nil, m, 0.05)
    local snap=g.CaptureManager:GetSnapshot(cap.id)
    assert(snap and snap.State=='Carrying', 'must be carrying after pickup duration')
    -- a carried pawn must stop running its own ground movement, on the server side too
    local lastMode = calls.modes and calls.modes[#calls.modes] or -1
    assert(lastMode==0, 'movement simulation must be stopped while carrying, got '..tostring(lastMode))
    assert(snap.PostId=='NEAR', 'nearest free post is claimed, got '..tostring(snap.PostId))
    assert(calls.attach and calls.attach.actor==m and calls.attach.socket=='headSocket')
    assert(calls.entered==1 and calls.invincible==true, 'frozen and invincible while carried')
    assert(m.HTCaptureState and m.HTCaptureState.PlayerKey==1 and m.HTCaptureState.Stage=='Carrying')
    -- the claim happens on the transition tick, the move request on the next one
    clock=clock+0.05; g.CaptureManager:Update(); t:ReceiveTickAI(nil, m, 0.05)
    assert((m.moves or 0)>=1, 'escort issued a move request')
    assert(m.lastMove and math.abs(m.lastMove.X-400)<1, 'move targets the post location')

    -- walk until arrival
    for i=1,200 do
        clock=clock+0.05
        walk(m, 0.05)
        g.CaptureManager:Update()
        t:ReceiveTickAI(nil, m, 0.05)
        if t.finished then break end
    end
    assert(t.finished==true, 'task must finish once the player is bound')
    assert(m.bb:GetValueAsObject('LockedTarget')==nil)
    assert(m.bb:GetValueAsBool('CatchReady')==false)
    assert(H.data(m).capture==nil)
    local post=g.CaptureManager.posts['NEAR']
    assert(calls.detach and calls.detach>=1, 'detached from the monster')
    assert(calls.attachComponent==post.actor.BindPoint, 'bound to the post bind point')
    assert(calls.relative and calls.relative.Z==90, 'bind height offset applied, got '..tostring(calls.relative and calls.relative.Z))
    assert(post.actor.PrisonState.PlayerKey==1 and post.actor.PrisonState.State=='Imprisoned')
    assert(post.actor.PrisonState.SessionId==cap.id,
        'prison payload must carry the session id so clients can accept a fresh session')
    assert(type(post.actor.PrisonState.StruggleWaitSeconds)=='number',
        'clients get a duration, not an absolute server timestamp')
    assert(m.HTCaptureState.Stage=='None' or m.HTCaptureState.PlayerKey==-1)
    assert(calls.entered==1, 'frozen state entered exactly once')
end)

-- ---------- release path ------------------------------------------------------
check('struggling free releases the player and frees the post', function()
    local g=freshWorld()
    local post=newPost('NEAR',400,0,0)
    assert(Service.RegisterPost(g,post))
    local m=newMonster(); local target=playerPawn(1, vec(50,0,88))
    local started,sessionId=Capture.begin(m, target)
    assert(started)
    imprison(g, sessionId, 'NEAR')
    local pc={key=1}
    local okBefore,why=Service.Struggle(g, post, pc)
    assert(not okBefore and why=='NOT_YET', 'struggle blocked before the grace period')
    clock=clock+H.Defaults.HTImprisonDuration+0.1
    g.CaptureManager:Update()
    for i=1,8 do
        clock=clock+1
        local okPress, count, need = Service.Struggle(g, post, pc)
        assert(okPress, 'press '..i..' refused: '..tostring(count))
        assert(need==8)
    end
    g.CaptureManager:Update()
    assert(g.CaptureManager:GetSnapshot(sessionId)==nil, 'session ended after 8 presses')
    assert(calls.left==1 and calls.invincible==false, 'frozen and invincible cleared')
    assert((calls.detach or 0)>=1)
    assert(calls.place and calls.placeTeleport==true, 'placed with bTeleport')
    assert(post.PrisonState.State=='Free' and post.PrisonState.PlayerKey==-1)
    assert(not Service.IsBusy(g,1))
end)

check('client attaches only its own pawn and detaches when the state clears', function()
    freshWorld()
    local monster=newMonster()
    local mine=playerPawn(1)
    calls={}
    UGCGameSystem.GetLocalPlayerController=function() return {K2_GetPawn=function() return mine end, key=1} end
    Presentation.Carrier(monster, {PlayerKey=1, Stage='Carrying', CarrySocket='headSocket', Revision=1})
    assert(calls.attach and calls.attach.actor==monster)
    calls={}
    Presentation.Carrier(monster, {PlayerKey=2, Stage='Carrying', Revision=2})
    assert(calls.detach==1, 'another player state must not keep mine attached')
    calls={}
    local post=newPost('NEAR',400,0,0)
    Presentation.Bind(post, {PlayerKey=1, State='Imprisoned', Revision=3})
    assert(calls.attachComponent==post.BindPoint)
    calls={}
    Presentation.Bind(post, {PlayerKey=-1, State='Free', Revision=4})
