local Manager=dofile(TEST_ROOT..'/Script/Gameplay/Desk/DeskSearchManager.lua')
local Config=dofile(TEST_ROOT..'/Script/Gameplay/Desk/DeskOutcomeConfig.lua')
local now,rolls=0,0
local distance,dead={},{}
local emitted={}
local m=Manager.new({now=function() return now end,
    roll=function() rolls=rolls+1; return 'Random'..rolls end,
    validate=function(_,p,r)
        if dead[p] then return false,'DEAD' end
        return (distance[p] or 0)<=r,'TOO_FAR'
    end,
    emit=function(a,s,k) emitted[#emitted+1]={s=s,k=k} end},Config)
local a,b={},{}
assert(m:Register(a,'A')); assert(m:Register(b,'B'))
assert(not m:Register({},'A'))
assert(not m:Register({},'Invalid',{CancelRadius=1}))
local ok,i,v=m:StartSearch('A',1); assert(ok)
assert(not m:StartSearch('A',2)); assert(not m:StartSearch('B',1))
assert(not m:CompleteSearch('A',v,i))
distance[1]=300; now=3; m:Update()
assert(m:GetSnapshot('A').State=='Closed'); assert(m:GetOpenedCount()==0)
distance[1]=0
ok,i,v=m:StartSearch('A',1); assert(ok)
assert(m:GetSnapshot('A').SearchEndTime==6)
assert(not m:CancelSearch('A',v,i-1,'old'))
now=6; assert(m:CompleteSearch('A',v,i)); assert(m:GetOpenedCount()==1)
assert(not m:CompleteSearch('A',v,i)); assert(m:GetOpenedCount()==1)
assert(m:GetSnapshot('A').FinalOutcomeId=='Random1')
assert(m:ResetDesk(a)); assert(m:GetOpenedCount()==1)
assert(not m:CompleteSearch('A',v,i))
for count=2,10 do
    assert(m:StartSearch('A',1)); now=now+3; m:Update()
    local result=m:GetSnapshot('A').FinalOutcomeId
    if count==5 then assert(result=='NoteEvent_05')
    elseif count==10 then assert(result=='NoteEvent_10')
    else assert(result:match('Random')) end
    if count<10 then assert(m:ResetDesk(a)) end
end
assert(m:GetOpenedCount()==10)
assert(m:ResetOpenedCount()); assert(m:GetOpenedCount()==0)
assert(m:GetSnapshot('A').State=='Opening')
assert(m:ResetAllDesks()); assert(m:GetOpenedCount()==0)
ok,i,v=m:StartSearch('A',1); assert(ok)
assert(m:ResetDesk(a)); assert(not m:CompleteSearch('A',v,i))
ok,i,v=m:StartSearch('A',1); assert(ok)
assert(not m:CancelSearch('A',v-1,i,'old'))
dead[1]=true; now=now+3; m:Update(); assert(m:GetOpenedCount()==0)
dead[1]=false
-- In-progress search survives count reset and commits in the new epoch.
assert(m:StartSearch('A',1)); m:ResetOpenedCount(); now=now+3; m:Update()
assert(m:GetOpenedCount()==1); assert(m:GetSnapshot('A').CounterEpoch==3)
-- Two completions crossing 5 select exactly one fixed outcome.
assert(m:ResetAllDesks()); m.TotalOpenedCount=4
assert(m:StartSearch('A',1)); assert(m:StartSearch('B',2))
now=now+3; m:Update()
local hits=0
for _,id in ipairs({'A','B'}) do
    if m:GetSnapshot(id).FinalOutcomeId=='NoteEvent_05' then hits=hits+1 end
end
assert(hits==1 and m:GetOpenedCount()==6)
assert(m:ResetAllDesks()); assert(m:StartSearch('A',1))
assert(m:Unregister(a)); assert(m.players[1]==nil)
assert(not m:ResetDesk(a)); assert(m:GetSnapshot('A')==nil)
assert(Config.Roll('Default',function() return 0 end)=='Ordinary')
assert(Config.Roll('Default',function() return 0.99 end)=='Strange')
assert(Config.Roll('Unknown')==nil)
-- Reentrancy is isolated: a consumer can reset after successful state is committed.
local reentrant
local resets=0
reentrant=Manager.new({now=function() return now end,
    roll=function() return 'Ordinary' end,validate=function() return true end,
    emit=function(actor,s,kind)
        if kind=='Opened' then resets=resets+1; reentrant:ResetDesk(actor) end
    end},Config)
assert(reentrant:Register(b,'R')); assert(reentrant:StartSearch('R',1))
now=now+3; reentrant:Update()
assert(resets==1 and reentrant:GetSnapshot('R').State=='Closed')
assert(reentrant:GetOpenedCount()==1)
-- Reset count re-arms fixed results while preserving a successful desk.
assert(reentrant:ResetOpenedCount())
for n=1,5 do
    assert(reentrant:StartSearch('R',1)); now=now+3; reentrant:Update()
end
assert(reentrant:GetOpenedCount()==5)
print('PASS: desk state machine, interruption, contention, 5/10 override, reset, epochs, stale sessions')
