local FocusState=dofile(TEST_ROOT..'/Script/Gameplay/Interaction/FocusState.lua')
local f=FocusState.new(7)
local a={DeskState={State='Closed'}}
local b={DeskState={State='Closed'}}
f:SetFocus(a); assert(f:CanRequest() and f:DisplayTarget()==a)
assert(f:ApplyDeskState(a,{State='Searching',SearchingPlayerId=7,Revision=2}))
assert(not f:CanRequest() and f:DisplayTarget()==a)
f:SetFocus(b); assert(f:DisplayTarget()==a and not f:CanRequest(), 'switching view must not move active search')
f:SetFocus(nil); assert(f:DisplayTarget()==a, 'looking away must not interrupt')
assert(not f:ApplyDeskState(a,{State='Closed',Revision=1}), 'old update must not clear current search')
assert(f:ApplyDeskState(a,{State='Closed',Revision=3}))
assert(f:DisplayTarget()==nil)
f:SetFocus(b); assert(f:CanRequest())
assert(f:ApplyDeskState(a,{State='Searching',SearchingPlayerId=8,Revision=4}))
assert(f:DisplayTarget()==b, 'other players searches must not own local progress')
assert(f:ApplyDeskState(b,{State='Searching',SearchingPlayerId=7,Revision=1}))
f:Remove(b); assert(f:DisplayTarget()==nil)
print('PASS: focus, active search, look-away, stale updates and teardown')
