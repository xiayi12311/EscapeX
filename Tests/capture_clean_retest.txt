local S=UGCGameSystem.UGCRequire('Script.Gameplay.Capture.CaptureService')
local gm=UGCGameSystem.GetGameMode()
local p=UGCGameSystem.GetAllPlayerPawn()[1]
local key=UGCGameSystem.GetPlayerKeyByPlayerPawn(p)
local pc=UGCGameSystem.GetPlayerControllerByPlayerKey(key)
local start=UGCGameSystem.GetTimeSeconds(gm)
local last,post,seen,bound,hits,minZ,maxGap=nil,nil,false,false,0,math.huge,0
if not S.IsBusy(gm,key) then
 local cls=UE.LoadClass('/EscapeX/Asset/Blueprint/Prefabs/Monsters/TheHeadTeacher.TheHeadTeacher_C')
 for _,m in pairs(UGCActorComponentUtility.GetAllActorsOfClass(p,cls)) do
  m.HTDamage=0
  local l=m:K2_GetActorLocation()
  local v=m:GetActorForwardVector()
  p:K2_SetActorLocation({X=l.X+v.X*100,Y=l.Y+v.Y*100,Z=l.Z},false,{},true)
  break
 end
end
local timer
timer=UGCTimerUtility.CreateLuaTimer(0.25,function()
 local s=gm.CaptureManager:SessionOfPlayer(key)
 local now=UGCGameSystem.GetTimeSeconds(gm)
 if s then
  seen=true
  if last~=s.state then
   last=s.state
   ugcprint('[HT_CLEAN] state='..s.state..' id='..s.id)
  end
  if s.state=='Imprisoned' then
   post=s.post.actor
   local l=p:K2_GetActorLocation()
   local b=post.BindPoint:K2_GetComponentLocation()
   local gap=math.abs(l.Z-(b.Z+post.BindHeightOffset))
   minZ=math.min(minZ,l.Z)
   maxGap=math.max(maxGap,gap)
   if not bound then
    bound=true
    ugcprint('[HT_CLEAN] BOUND z='..l.Z..' postZ='..b.Z..' move='..tostring(UGCPlayerPawnSystem.AllowPawnState(p,EPawnState.Move))..' fire='..tostring(UGCPlayerPawnSystem.AllowPawnState(p,EPawnState.GunFire))..' imp='..tostring(UGCPlayerPawnSystem.HasPawnState(p,EPawnState.Imprisonment)))
    local ok,reason=S.Struggle(gm,post,pc)
    ugcprint('[HT_CLEAN] early='..tostring(ok)..' reason='..tostring(reason))
   end
   if s.ready then
    local ok,count,need=S.Struggle(gm,post,pc)
    if ok then hits=hits+1 end
    ugcprint('[HT_CLEAN] press='..tostring(ok)..' count='..tostring(count)..' need='..tostring(need))
   end
  end
 elseif bound then
  UGCTimerUtility.RemoveLuaTimer(timer)
  ugcprint('[HT_CLEAN] DONE hits='..hits..' minZ='..minZ..' maxBindZError='..maxGap..' postSession='..tostring(gm.CaptureManager:FindPost(post).session)..' move='..tostring(UGCPlayerPawnSystem.AllowPawnState(p,EPawnState.Move))..' fire='..tostring(UGCPlayerPawnSystem.AllowPawnState(p,EPawnState.GunFire))..' imp='..tostring(UGCPlayerPawnSystem.HasPawnState(p,EPawnState.Imprisonment))..' invincible='..tostring(UGCPlayerPawnSystem.GetIsInvincible(p)))
 elseif seen then
  UGCTimerUtility.RemoveLuaTimer(timer)
  ugcprint('[HT_CLEAN] FAIL ended before bind')
 end
 if now-start>90 then UGCTimerUtility.RemoveLuaTimer(timer);ugcprint('[HT_CLEAN] FAIL timeout')end
end,true,'HT_CLEAN')
