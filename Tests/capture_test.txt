-- Offline rules for the capture/imprison state machine. No UE replication, no
-- navigation, no attachment: those are covered by the PIE checklist in
-- Docs/TheHeadTeacher_抓住囚禁_实现计划.md section 6.
local Manager=dofile(TEST_ROOT..'/Script/Gameplay/Capture/CaptureManager.lua')
local Config=dofile(TEST_ROOT..'/Script/Gameplay/Capture/CaptureConfig.lua')

local now,dead,rescuerOK
local emitted

local function newManager()
    emitted={}
    return Manager.new({
        now=function() return now end,
        validatePlayer=function(key)
            if dead[key] then return false,'PLAYER_DEAD' end
            return true
        end,
        validateRescuer=function(post,key,radius)
            if not rescuerOK[key] then return false,'RESCUER_TOO_FAR' end
            return true
        end,
        emit=function(data,kind,reason,post,monster)
            emitted[#emitted+1]={k=kind,r=reason,data=data,post=post,monster=monster}
        end,
        log=function() end,
    },Config)
end

local sessionOptions={PickupDuration=0.6,PutDownDuration=0.6,EscortTimeout=20,
    ImprisonDuration=15,StruggleInterval=0.15,EscortSpeed=300,ArriveRadius=120}
local postOptions={StrugglePresses=3,RescueDuration=3,RescueRadius=200}

local function kinds()
    local list={}
    for _,e in ipairs(emitted) do list[#list+1]=e.k end
    return table.concat(list,',')
end

local function kindAt(kind)
    local index,found
    for i,e in ipairs(emitted) do if e.k==kind then index,found=i,e end end
    return index,found                                          -- last match wins
end

-- ---- captured player leaves in every live state ------------------------------
for _,state in ipairs({'Grabbing','Carrying','PuttingDown','Imprisoned'}) do
    now=0; dead={}; rescuerOK={}
    m=newManager()
    assert(m:RegisterPost({},'A',{Location={X=0,Y=0,Z=0}}))
    local idX
    ok,idX=m:StartSession(31,{},sessionOptions,postOptions)
    assert(ok)
    if state~='Grabbing' then
        now=0.7; m:Update(); assert(m:ClaimPost(idX,'A'))
    end
    if state=='PuttingDown' or state=='Imprisoned' then
        m:MarkArrived(idX); now=1; m:Update()
    end
    if state=='Imprisoned' then now=2; m:Update() end
    assert(m:GetSession(idX).state==state)
    emitted={}
    dead[31]=true; now=now+0.1; m:Update()
    local _,gone=kindAt('Released')
    assert(gone and gone.r=='PLAYER_DEAD')
    assert(not m:IsBusy(31))
    assert(#m:NearestFreePosts({X=0,Y=0,Z=0})==1)
end

-- an emit callback may cancel the session it is told about
local reentrant
reentrant=Manager.new({now=function() return now end,
    validatePlayer=function() return true end,
    validateRescuer=function() return true end,
    emit=function(data,kind,reason,post,monster)
        if kind=='Imprisoned' then reentrant:CancelPlayer(data.PlayerKey,'CANCELLED') end
    end},Config)
assert(reentrant:RegisterPost({},'A',{Location={X=0,Y=0,Z=0}}))
local idRe
now=0
ok,idRe=reentrant:StartSession(61,{},sessionOptions,postOptions)
assert(ok)
assert(reentrant:ClaimPost(idRe,'A'))
now=0.7; reentrant:Update(); reentrant:MarkArrived(idRe)
now=1; reentrant:Update(); now=2; reentrant:Update()
assert(not reentrant:GetSession(idRe))
assert(#reentrant:NearestFreePosts({X=0,Y=0,Z=0})==1)
reentrant:FlushEvents()                                   -- Released was queued
local queue=reentrant.events
assert(#queue==0)
