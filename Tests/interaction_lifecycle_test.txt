-- Wiring test: who owns the focus layer, and does every lifecycle edge reach it exactly
-- once. Mocks only; the native side is verified separately in PIE.
--
-- Ownership decided by measurement (PIE 2026-10-27): the local player controller arms the
-- monitor, because pawn BeginPlay/EndPlay ordering across a pawn replacement is not
-- dependable - the replacing pawn never started a session and the old pawn's EndPlay tore
-- the running layer down silently.
local calls={monitor=0,begin=0,stops=0,changed=0,removed=0,deskCancel=0,captureCancel=0,registerEnd=0}
local focusFake={
    Monitor=function() calls.monitor=calls.monitor+1; return true end,
    Begin=function(pawn) calls.begin=calls.begin+1; calls.beginPawn=pawn; return true end,
    End=function() calls.stops=calls.stops+1 end,
    DeskChanged=function(actor,state) calls.changed=calls.changed+1; calls.lastActor=actor; calls.lastState=state end,
    DeskRemoved=function(actor) calls.removed=calls.removed+1 end,
}
local serviceFake={Unregister=function() calls.registerEnd=calls.registerEnd+1 end,
    Register=function() return true end,
    StartSearch=function() return true end,
    CancelPlayer=function() calls.deskCancel=calls.deskCancel+1 end}
local captureFake={CancelPlayer=function() calls.captureCancel=calls.captureCancel+1 end}
local presentationFake={Apply=function() end,Begin=function() end,End=function() end}

local stubs={
    ['Script.Gameplay.Interaction.DeskFocusPresentation']=focusFake,
    ['Script.Gameplay.Desk.DeskSearchService']=serviceFake,
    ['Script.Gameplay.Capture.CaptureService']=captureFake,
    ['Script.Gameplay.Desk.DeskPresentation']=presentationFake,
    ['Script.Gameplay.Desk.DeskOutcomeConfig']={},
    ['Script.Gameplay.Interaction.InteractionConfig']={},
    ['Script.Gameplay.Interaction.DeskClock']={},
    ['Script.Gameplay.Interaction.FocusState']={},
}
KismetSystemLibrary={IsValid=function(o) return o~=nil and not o.destroyed end}
UGCTimerUtility={CreateLuaTimer=function() return 't' end,
    RemoveLuaTimer=function(handle) if handle=='t' then calls.registerTimerCleared=true end end}
UGCGameSystem={UGCRequire=function(name)
    local m=stubs[name]
    assert(m,'unexpected require in lifecycle test: '..tostring(name))
    return m
end, GetGameMode=function() return {} end}

-- ---- pawn file must stay inert ----
local pawn=dofile(TEST_ROOT..'/Script/Blueprint/UGCPlayerPawn.lua')
assert(pawn.ReceiveBeginPlay==nil,'the pawn must not drive the focus layer')
assert(pawn.ReceiveEndPlay==nil,'the pawn must not tear the focus layer down')

-- ---- controller arms the monitor and still clears sessions on disconnect ----
local controller=dofile(TEST_ROOT..'/Script/Blueprint/UGCPlayerController.lua')
controller.SuperClass={ReceiveBeginPlay=function() calls.superBegin=(calls.superBegin or 0)+1 end,
    UGC_PlayerLostConnectionEvent=function() calls.superLost=(calls.superLost or 0)+1 end}
controller.ReceiveBeginPlay({})
assert(calls.monitor==1,'the local controller must arm the focus monitor')
assert(calls.superBegin==1,'the controller must chain into its superclass')
assert(calls.begin==0,'arming the monitor must not start a session directly')

local conn={HasAuthority=function() return true end}
setmetatable(conn,{__index=controller})
conn.UGC_PlayerLostConnectionEvent(conn)
assert(calls.deskCancel==1 and calls.captureCancel==1,'disconnect must clear both sessions')
assert(calls.superLost==1,'disconnect must chain into its superclass')

-- ---- desk replication and teardown reach the focus layer ----
local desk=dofile(TEST_ROOT..'/Script/Blueprint/Prefabs/Interactables/BP_SearchableDesk.lua')
desk.SuperClass={ReceiveBeginPlay=function() end,ReceiveEndPlay=function() end}
local instance={DeskState={State='Searching',Revision=3},HasAuthority=function() return false end}
desk.OnRep_DeskState(instance)
assert(calls.changed==1,'a replicated desk state must reach the focus layer')
assert(calls.lastActor==instance and calls.lastState==instance.DeskState)
instance.destroyed=true
desk.ReceiveEndPlay(instance,'Reason')
assert(calls.removed==1,'a destroyed desk must be dropped by the focus layer')

-- server-side teardown must not touch the client-only focus layer, and must release
-- the registration timer it owns
local serverDesk={HasAuthority=function() return true end,DeskRegisterTimer='t'}
desk.ReceiveEndPlay(serverDesk,'Reason')
assert(calls.removed==1,'server teardown must not remove a client focus entry')
assert(calls.registerTimerCleared,'the register timer must be released on teardown')
print('PASS: controller owns the focus layer; desk events and teardown reach it exactly once')
